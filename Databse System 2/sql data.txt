-- 1. Create and select the database
CREATE DATABASE IF NOT EXISTS steamdb;
USE steamdb;

-- 2. Drop tables if they exist
DROP TABLE IF EXISTS TBOrderDetails;
DROP TABLE IF EXISTS TBOrders;
DROP TABLE IF EXISTS TBLibrary;
DROP TABLE IF EXISTS TBActivityLog;
DROP TABLE IF EXISTS TBGames;
DROP TABLE IF EXISTS TBPublishers;
DROP TABLE IF EXISTS TBUsers;

-- 3. Create Tables
CREATE TABLE TBPublishers (
    PublisherID   INT AUTO_INCREMENT PRIMARY KEY,
    PublisherName VARCHAR(100) NOT NULL,
    SupportEmail  VARCHAR(100),
    Country       VARCHAR(50)
);

CREATE TABLE TBUsers (
    UserID         INT AUTO_INCREMENT PRIMARY KEY,
    Username       VARCHAR(50)  NOT NULL,
    Email          VARCHAR(100) NOT NULL,
    Password       VARCHAR(255) NOT NULL,
    Role           VARCHAR(20) DEFAULT 'user',
    WalletBalance  DECIMAL(10,2) DEFAULT 0.00,
    CreatedAt      TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    AccountCreated DATE
);

CREATE TABLE TBGames (
    GameID            INT AUTO_INCREMENT PRIMARY KEY,
    PublisherID       INT,
    Title             VARCHAR(200) NOT NULL,
    CoverImage        VARCHAR(500),
    Genre             VARCHAR(50),
    Price             DECIMAL(10,2),
    ReleaseDate       DATE,
    StorageRequirement INT,
    FOREIGN KEY (PublisherID) REFERENCES TBPublishers(PublisherID)
);

CREATE TABLE TBOrders (
    OrderID       INT AUTO_INCREMENT PRIMARY KEY,
    UserID        INT,
    OrderDate     DATE,
    TotalAmount   DECIMAL(10,2),
    PaymentMethod VARCHAR(50),
    FOREIGN KEY (UserID) REFERENCES TBUsers(UserID)
);

CREATE TABLE TBOrderDetails (
    OrderDetailID INT AUTO_INCREMENT PRIMARY KEY,
    OrderID       INT,
    GameID        INT,
    PricePaid     DECIMAL(10,2),
    FOREIGN KEY (OrderID) REFERENCES TBOrders(OrderID),
    FOREIGN KEY (GameID)  REFERENCES TBGames(GameID)
);

CREATE TABLE TBLibrary (
    LibraryID     INT AUTO_INCREMENT PRIMARY KEY,
    UserID        INT,
    GameID        INT,
    AcquiredDate  DATE,
    PlaytimeHours DECIMAL(10,1),
    FOREIGN KEY (UserID) REFERENCES TBUsers(UserID),
    FOREIGN KEY (GameID) REFERENCES TBGames(GameID)
);

CREATE TABLE TBActivityLog (
    LogID      INT AUTO_INCREMENT PRIMARY KEY,
    UserID     INT,
    Action     VARCHAR(50),
    EntityType VARCHAR(50),
    EntityID   INT,
    Details    VARCHAR(255),
    CreatedAt  TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
